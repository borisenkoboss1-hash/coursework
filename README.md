# Дверь.рф — Курсовая работа

## Структура проекта

```
dveri-rf/
├── index.html          ← Главная страница
├── catalog.html        ← Каталог (фильтры, сортировка, пагинация)
├── product.html        ← Карточка товара (галерея, вкладки)
├── cart.html           ← Корзина (промокоды, оформление заказа)
├── favorites.html      ← Избранное
├── register.html       ← Регистрация с валидацией
├── login.html          ← Вход в аккаунт
├── profile.html        ← Личный кабинет пользователя
├── admin.html          ← Панель администратора
├── about.html          ← О компании
├── delivery.html       ← Доставка и установка
├── accessible.html     ← Версия для слабовидящих
│
├── css/
│   ├── style.css       ← Общие стили (все страницы)
│   ├── index.css       ← Главная страница
│   ├── catalog.css     ← Каталог
│   ├── product.css     ← Карточка товара
│   ├── cart.css        ← Корзина и избранное
│   ├── auth.css        ← Регистрация и вход
│   ├── profile.css     ← Личный кабинет
│   ├── admin.css       ← Панель администратора
│   ├── pages.css       ← О нас и Доставка
│   └── accessible.css  ← Версия для слабовидящих
│
├── js/
│   ├── common.js       ← Общий JS (тема, язык, бургер, вход)
│   ├── index.js        ← Слайдер, секции товаров на главной
│   ├── catalog.js      ← Фильтры, сортировка, пагинация
│   ├── product.js      ← Галерея, вкладки, похожие товары
│   ├── cart.js         ← Корзина, промокоды, заказ
│   ├── favorites.js    ← Избранное
│   ├── register.js     ← Валидация регистрации
│   ├── login.js        ← Вход в аккаунт
│   ├── profile.js      ← Личный кабинет
│   ├── admin.js        ← Панель администратора
│   └── accessible.js  ← Версия для слабовидящих
│
└── data/
    ├── products.json   ← 32 товара
    ├── users.json      ← 11 пользователей (1 admin + 10 user)
    ├── cart.json       ← Тестовые корзины
    └── favorites.json  ← Тестовое избранное
```
